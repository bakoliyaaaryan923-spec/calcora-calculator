function calculate(){ if(!current) return; try{ const old = current.trim(); // ========================================== // VERY LARGE INTEGER CALCULATIONS // Uses BigInt to prevent precision loss // ========================================== if( /^[0-9+\-*()\s]+$/.test(old) && !old.includes("/") && !old.includes(".") ){ const numbersOnly = old.replace(/[^0-9]/g,""); if(numbersOnly.length > 15){ const bigExpression = old.replace( /(\d+)/g, "$1n" ); const result = Function( '"use strict"; return (' + bigExpression + ')' )(); current = result.toString(); expression.textContent = old + " ="; update(); addHistory( old, current ); return; } } // ========================================== // NORMAL CALCULATIONS // ========================================== const safe = old.replace( /[^0-9+\-*/().]/g, "" ); if(!safe){ throw new Error(); } const result = Function( '"use strict"; return (' + safe + ')' )(); if(!Number.isFinite(result)){ throw new Error(); } current = String( Number(result.toFixed(12)) ); expression.textContent = old + " ="; update(); addHistory( old, current ); }catch(error){ current = "Error"; expression.textContent = ""; update(); } }
